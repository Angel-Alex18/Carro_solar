    const mesh = MeshBuilder.CreateBox(name, {
      width: w, height: h, depth: d,
      ...(opts.bevel ? { bevel: true } : {})
    }, scene);
    mesh.position.copyFrom(pos);
    mesh.material = mat;
    mesh.parent = parent;
    return mesh;
  }

    const mesh = MeshBuilder.CreateCylinder(name, {
      diameter, height, tessellation: tess
    }, scene);
    mesh.position.copyFrom(pos);
    mesh.material = mat;
    mesh.parent = parent;
    return mesh;
  }
  function sphere(parent, name, diameter, pos, mat) {
    const mesh = MeshBuilder.CreateSphere(name, { diameter, segments: 12 }, scene);
    mesh.position.copyFrom(pos);
    mesh.material = mat;
    mesh.parent = parent;
    return mesh;
  }

    const mesh = MeshBuilder.CreateTube(name, {
      path: points.map(p => new Vector3(p[0], p[1], p[2])),
      radius, tessellation: 10, cap: Mesh.CAP_ALL

    mesh.parent = parent;
    return mesh;

function addRealisticBehaviors(scene, pieces) {
  const updaters = [];
  const clickHandlers = {};
  const flashState = new Map();
  const globalState = { powerOn: true };

  scene.onBeforeRenderObservable.add(() => {
    const dt = Math.min(scene.getEngine().getDeltaTime() / 1000, 0.05);
    for (let i = updaters.length - 1; i >= 0; i--) {
      if (updaters[i](dt) === false) updaters.splice(i, 1);
    }
    const toDel = [];
    flashState.forEach((s, mesh) => {
      s.t += dt;
      mesh.overlayAlpha = s.maxA * Math.max(0, 1 - s.t / s.dur);
      if (s.t >= s.dur) { mesh.renderOverlay = false; toDel.push(mesh); }
    });
    toDel.forEach(m => flashState.delete(m));
  });

  pieces.forEach(p => p.getChildMeshes().forEach(m => m.isPickable = true));

  scene.onPointerObservable.add((pi) => {
    if (pi.type !== PointerEventTypes.POINTERPICK) return;
    const pick = scene.pick(scene.pointerX, scene.pointerY, m => m.isPickable);
    if (!pick || !pick.pickedMesh) return;
    let node = pick.pickedMesh;
    while (node && !pieces.has(node.name)) node = node.parent;
    if (!node) return;
    const h = clickHandlers[node.name];
    if (h) h(node, pick.pickedMesh);
  });

  const child = (piece, name) => piece.getChildMeshes().find(m => m.name === name);

  const flash = (mesh, color, dur = 0.8, maxA = 0.6) => {
    if (!mesh) return;
    mesh.renderOverlay = true;
    mesh.overlayColor = color;
    flashState.set(mesh, { dur, maxA, t: 0 });
    mesh.overlayAlpha = maxA;
  };

  const makeDisplay = (parent, position, w = 0.68, h = 0.4) => {
    const tex = new DynamicTexture("dt_" + Math.random().toString(36).slice(2), { width: 256, height: 128 }, scene, true);
    tex.hasAlpha = true;
    const mat = new StandardMaterial("dtm_" + Math.random().toString(36).slice(2), scene);
    mat.diffuseTexture = tex;
    mat.emissiveTexture = tex;
    mat.emissiveColor = new Color3(0.2, 1, 0.5);
    mat.disableLighting = true;
    mat.backFaceCulling = false;
    const plane = MeshBuilder.CreatePlane("display", { width: w, height: h }, scene);
    plane.material = mat;
    plane.parent = parent;
    plane.position.copyFrom(position);
    plane.billboardMode = Mesh.BILLBOARDMODE_ALL;
    plane.isPickable = false;
    return { plane, tex, mat };
  };

  const drawText = (disp, text) => {
    disp.tex.getContext().clearRect(0, 0, 256, 128);
    disp.tex.drawText(text, null, 80, "bold 56px monospace", "#00ff88", "transparent", true);
  };

  let drillOn = false;
  let welding = false;

  // Jaula
  clickHandlers["jaula_seguridad"] = (p) => {
    p.getChildMeshes().forEach(m => flash(m, new Color3(0.1, 0.9, 0.3), 1.2, 0.5));
  };

  // Arnés
  clickHandlers["arnes_cuatro_puntos"] = (p) => {
    flash(child(p, "hebilla"), new Color3(0, 1, 0), 0.6);
    flash(child(p, "boton_hebilla"), new Color3(1, 0.3, 0), 0.6);
  };

  // Habitáculo
  clickHandlers["proteccion_habitaculo"] = (p) => {
    p.getChildMeshes().forEach(m => flash(m, new Color3(0.2, 0.5, 1), 0.8));
  };

  // Corte de emergencia
  {
    const corte = pieces.get("corte_emergencia");
    const boton = child(corte, "boton_rojo");
    const y0 = boton.position.y;
    clickHandlers["corte_emergencia"] = () => {
      let t = 0;
      updaters.push((dt) => {
        t += dt;
        const ph = t < 0.12 ? t / 0.12 : t < 0.24 ? 1 - (t - 0.12) / 0.12 : 0;
        boton.position.y = y0 - ph * 0.08;
        if (t > 0.24) { boton.position.y = y0; return false; }
      });
      globalState.powerOn = !globalState.powerOn;
      const color = globalState.powerOn ? new Color3(0, 1, 0) : new Color3(1, 0, 0);
      pieces.forEach(p => {
        if (p.name === "corte_emergencia") return;
        p.getChildMeshes().forEach(m => flash(m, color, 0.6, 0.35));
      });
      if (!globalState.powerOn) { drillOn = false; welding = false; }
      console.log(globalState.powerOn ? "✅ Sistema ON" : "🚨 PARO DE EMERGENCIA — OFF");
    };
  }

  // Extintor
  {
    const ext = pieces.get("extintor_senalizacion");
    const nozzle = new TransformNode("nozzle", scene);
    nozzle.parent = ext;
    nozzle.position.set(0.08, 1.35, 0);
    const spray = new ParticleSystem("spray", 600, scene);
    spray.particleTexture = new Texture("https://assets.babylonjs.com/textures/flare.png", scene);
    spray.emitter = nozzle;
    spray.minEmitBox = new Vector3(-0.05, -0.05, -0.05);
    spray.maxEmitBox = new Vector3(0.05, 0.05, 0.05);
    spray.color1 = new Color4(1, 1, 1, 0.8);
    spray.color2 = new Color4(0.7, 0.9, 1, 0.6);
    spray.colorDead = new Color4(0.8, 0.8, 0.9, 0);
    spray.minSize = 0.05; spray.maxSize = 0.25;
    spray.minLifeTime = 0.3; spray.maxLifeTime = 0.9;
    spray.emitRate = 0;
    spray.blendMode = ParticleSystem.BLENDMODE_STANDARD;
    spray.gravity = new Vector3(0, -2, 0);
    spray.direction1 = new Vector3(-1, 0.5, -0.5);
    spray.direction2 = new Vector3(-3, 2, 0.5);
    spray.minEmitPower = 0.5; spray.maxEmitPower = 2;
    spray.updateSpeed = 0.02;
    spray.start();
    clickHandlers["extintor_senalizacion"] = () => {
      spray.emitRate = 400;
      let t = 0;
      updaters.push((dt) => { t += dt; if (t > 1.2) { spray.emitRate = 0; return false; } });
      console.log("🧯 Descarga de extintor");
    };
  }

  // Multímetro
  {
    const multi = pieces.get("multimetro");
    const disp = makeDisplay(multi, new Vector3(0, 1.15, 0.13), 0.68, 0.4);
    drawText(disp, "0.00 V");
    const sel = child(multi, "selector");
    let selAngle = 0, ri = 0;
    const readings = ["0.00 V", "12.4 V", "230 V", "0.85 Ω", "12.6 mA", "5.02 V"];
    clickHandlers["multimetro"] = () => {
      ri = (ri + 1) % readings.length;
      drawText(disp, readings[ri]);
      selAngle -= Math.PI / 6;
    };
    updaters.push((dt) => {
      if (sel) sel.rotation.y += (selAngle - sel.rotation.y) * Math.min(1, dt * 8);
    });
  }

  // Pinza amperimétrica
  {
    const pinza = pieces.get("pinza_amperimetrica");
    const disp = makeDisplay(pinza, new Vector3(0, 0.78, 0.13), 0.5, 0.28);
    drawText(disp, "0.00 A");
    const mordaza = child(pinza, "mordaza");
    const y0 = mordaza.position.y;
    let opened = false, targetY = 0, ri = 0;
    const readings = ["0.00 A", "1.25 A", "3.80 A", "12.4 A", "0.45 A"];
    clickHandlers["pinza_amperimetrica"] = () => {
      opened = !opened;
      targetY = opened ? -0.15 : 0;
      ri = (ri + 1) % readings.length;
      drawText(disp, opened ? readings[ri] : "--- A");
    };
    updaters.push((dt) => {
      const curr = mordaza.position.y - y0;
      mordaza.position.y = y0 + curr + (targetY - curr) * Math.min(1, dt * 8);
    });
  }

  // Soldadura
  {
    const sold = pieces.get("equipo_soldadura");
    const porta = child(sold, "porta_electrodo");
    const spark = new ParticleSystem("spark", 800, scene);
    spark.particleTexture = new Texture("https://assets.babylonjs.com/textures/flare.png", scene);
    spark.emitter = porta;
    spark.minEmitBox = new Vector3(-0.02, -0.02, -0.02);
    spark.maxEmitBox = new Vector3(0.02, 0.02, 0.02);
    spark.color1 = new Color4(1, 0.9, 0.3, 1);
    spark.color2 = new Color4(1, 0.5, 0, 1);
    spark.colorDead = new Color4(0.5, 0, 0, 0);
    spark.minSize = 0.02; spark.maxSize = 0.07;
    spark.minLifeTime = 0.2; spark.maxLifeTime = 0.7;
    spark.emitRate = 0;
    spark.blendMode = ParticleSystem.BLENDMODE_ONEONE;
    spark.gravity = new Vector3(0, -9, 0);
    spark.direction1 = new Vector3(-1.5, 0, -1.5);
    spark.direction2 = new Vector3(1.5, 3, 1.5);
    spark.minEmitPower = 0.5; spark.maxEmitPower = 3;
    spark.updateSpeed = 0.02;
    spark.start();
    const arc = new PointLight("arc", new Vector3(0, 0, 0), scene);
    arc.parent = porta;
    arc.diffuse = new Color3(1, 0.7, 0.3);
    arc.intensity = 0;
    arc.range = 2.5;
    let ft = 0;
    updaters.push((dt) => {
      spark.emitRate = (welding && globalState.powerOn) ? 250 : 0;
      if (welding && globalState.powerOn) {
        ft += dt;
        arc.intensity = 1 + Math.sin(ft * 40) * 0.5 + Math.random() * 0.3;
      } else { arc.intensity = 0; }
    });
    clickHandlers["equipo_soldadura"] = () => {
      if (!globalState.powerOn) { console.log("⚠️ Sin energía"); return; }
      welding = !welding;
      console.log(welding ? "⚡ Soldadura ON" : "⚡ Soldadura OFF");
    };
  }

  // Crimpadora
  clickHandlers["crimpadora_terminales"] = (p) => {
    const mi = child(p, "mandibula_izq");
    const md = child(p, "mandibula_der");
    const x0i = mi.position.x, x0d = md.position.x;
    let t = 0;
    updaters.push((dt) => {
      t += dt;
      const k = t < 0.25 ? t / 0.25 : 1 - (t - 0.25) / 0.25;
      mi.position.x = x0i + k * 0.08;
      md.position.x = x0d - k * 0.08;
      if (t > 0.5) { mi.position.x = x0i; md.position.x = x0d; return false; }
    });
  };

  // Llaves / destornilladores
  clickHandlers["juego_llaves_destornilladores"] = (p) => {
    const llaves = p.getChildMeshes().filter(m => m.name.startsWith("llave_mango"));
    llaves.forEach((m, i) => {
      const y0 = m.rotation.y;
      let t = -i * 0.1;
      updaters.push((dt) => {
        t += dt;
        if (t < 0) return;
        const ang = Math.min(t, 1);
        m.rotation.y = y0 + ang * Math.PI * 2;
        if (ang >= 1) { m.rotation.y = y0; return false; }
      });
    });
  };

  // Taladro
  {
    const tal = pieces.get("taladro_brocas");
    const broca = child(tal, "broca_principal");
    const cuerpo = child(tal, "cuerpo_taladro");
    const cPos = cuerpo.position.clone();
    let vt = 0;
    updaters.push((dt) => {
      if (drillOn && globalState.powerOn) {
        vt += dt;
        broca.rotation.y += dt * 30;
        cuerpo.position.x = cPos.x + Math.sin(vt * 80) * 0.008;
        cuerpo.position.y = cPos.y + Math.cos(vt * 90) * 0.006;
      } else {
        cuerpo.position.copyFrom(cPos);
      }
    });
    clickHandlers["taladro_brocas"] = () => {
      if (!globalState.powerOn) { console.log("⚠️ Sin energía"); return; }
      drillOn = !drillOn;
      console.log(drillOn ? "🔩 Taladro ON" : "🔩 Taladro OFF");
    };
  }

  // EPP (ciclo automático)
  {
    const epp = pieces.get("equipo_proteccion_personal");
    const parts = ["casco", "lente_izq", "lente_der", "guante_izq", "guante_der", "bota_izq", "bota_der"];
    let idx = 0, timer = 0;
    updaters.push((dt) => {
      timer += dt;
      if (timer > 2.5) {
        timer = 0;
        const mesh = child(epp, parts[idx % parts.length]);
        flash(mesh, new Color3(0.3, 0.9, 1), 1.2, 0.4);
        idx++;
      }
    });
    clickHandlers["equipo_proteccion_personal"] = (p, m) => {
      flash(m, new Color3(1, 0.9, 0.2), 1.0, 0.7);
    };
  }

  // HUD
  if (typeof document !== "undefined") {
    const old = document.getElementById("babylon-hud");
    if (old) old.remove();
    const hud = document.createElement("div");
    hud.id = "babylon-hud";
    hud.style.cssText = "position:fixed;bottom:10px;left:10px;color:#fff;font-family:sans-serif;font-size:13px;background:rgba(0,0,0,0.6);padding:8px 12px;border-radius:6px;z-index:9999;pointer-events:none;line-height:1.4;";
    hud.innerHTML = "🖱️ <b>Click en cada pieza</b> para interactuar<br>🚨 Paro de emergencia apaga todo el sistema";
    document.body.appendChild(hud);
  }

  return { updaters, clickHandlers, globalState };
}

// ─────────────────────────────────────────────────────────────
// 3) ESCENA
// ─────────────────────────────────────────────────────────────
function createScene() {
  const scene = new BABYLON.Scene(engine);

  const camera = new BABYLON.ArcRotateCamera(
    "camera",
    -Math.PI / 2,
    Math.PI / 3,
    22,
    BABYLON.Vector3.Zero(),
    scene
  );
  camera.setTarget(new BABYLON.Vector3(6.3, 0.5, 4.2));
  camera.attachControl(canvas, true);
  camera.wheelDeltaPercentage = 0.02;

  new BABYLON.HemisphericLight("light", new BABYLON.Vector3(0, 1, 0), scene);
  const dir = new BABYLON.DirectionalLight("dir", new BABYLON.Vector3(-1, -2, -1), scene);
  dir.position = new BABYLON.Vector3(20, 30, 20);

  const pieces = createSafetyAndTools(scene);
  addRealisticBehaviors(scene, pieces);

  return scene;
}

export default createScene;
